	// Every unsuccessful exit releases the partially constructed device/resources.

	template <typename Cleanup>

	class Rollback

	public:

		~Rollback() { if (!m_complete) m_cleanup(); }

		Rollback(const Rollback&) = delete;

		Rollback& operator=(const Rollback&) = delete;

		void Complete() { m_complete = true; }

	private:

		Cleanup m_cleanup;

		bool m_complete = false;

	};



	// Quality zero is valid only if both formats advertise at least one level.

	template <typename Query>

	D3DMULTISAMPLE_TYPE SelectMultisampling(bool enabled, D3DFORMAT color, D3DFORMAT depth, Query query)

	{

		if (!enabled) return D3DMULTISAMPLE_NONE;

		for (auto samples : {D3DMULTISAMPLE_4_SAMPLES, D3DMULTISAMPLE_2_SAMPLES})

		{

			DWORD colorLevels = 0, depthLevels = 0;

			if (SUCCEEDED(query(color, samples, &colorLevels)) && colorLevels &&

				SUCCEEDED(query(depth, samples, &depthLevels)) && depthLevels)

				return samples;

		}

		return D3DMULTISAMPLE_NONE;

	}



	inline DWORD BestFilter(DWORD caps, DWORD anisotropic, DWORD linear)

	{

		return (caps & anisotropic) ? D3DTEXF_ANISOTROPIC :

			((caps & linear) ? D3DTEXF_LINEAR : D3DTEXF_POINT);

	}

			Tracen("IDirect3DDevice.CreateDevice - ERROR D3DERR_NOTAVAILABLE\nThis device does not support the queried technique. ");
			break;
		case D3DERR_OUTOFVIDEOMEMORY:
			Tracen("IDirect3DDevice.CreateDevice - ERROR D3DERR_OUTOFVIDEOMEMORY\nDirect3D does not have enough display memory to perform the operation");
			break;
		default:
			Tracenf("IDirect3DDevice.CreateDevice - ERROR %d", ms_hLastResult);
			break;
		}

		if (ErrorCorrection)
			return CREATE_DEVICE;

		// Retry with conservative settings
		disableMSAA = true;
		iReflashRate = 0;
		++ErrorCorrection;
		iRet = CREATE_REFRESHRATE;
		goto RETRY;
	}

	// Check DXT Support Info
	const D3DFORMAT baseFormatForTextureCheck = Windowed ? d3dDisplayMode.Format : ms_d3dPresentParameter.BackBufferFormat;
	if (ms_lpd3d->CheckDeviceFormat(
		D3DADAPTER_DEFAULT,
		D3DDEVTYPE_HAL,
		baseFormatForTextureCheck,
		0,
		D3DRTYPE_TEXTURE,
		D3DFMT_DXT1) == D3DERR_NOTAVAILABLE)
	{
		ms_bSupportDXT = false;
	}

	if (ms_lpd3d->CheckDeviceFormat(
		D3DADAPTER_DEFAULT,
		D3DDEVTYPE_HAL,
		baseFormatForTextureCheck,
		0,
		D3DRTYPE_TEXTURE,
		D3DFMT_DXT3) == D3DERR_NOTAVAILABLE)
	{
		ms_bSupportDXT = false;
	}

	if (ms_lpd3d->CheckDeviceFormat(
		D3DADAPTER_DEFAULT,
		D3DDEVTYPE_HAL,
		baseFormatForTextureCheck,
		0,
		D3DRTYPE_TEXTURE,
		D3DFMT_DXT5) == D3DERR_NOTAVAILABLE)
	{
		ms_bSupportDXT = false;
	}

	if (FAILED((ms_hLastResult = ms_lpd3dDevice->GetDeviceCaps(&ms_d3dCaps))))
	{
		Tracenf("IDirect3DDevice.GetDeviceCaps - ERROR %d", ms_hLastResult);
		return CREATE_GET_DEVICE_CAPS2;
	}

	if (!Windowed)
		SetWindowPos(hWnd, HWND_TOPMOST, 0, 0, iHres, iVres, SWP_SHOWWINDOW);

	//Tracef("vertex shader version : %X\n",(DWORD)ms_d3dCaps.VertexShaderVersion);

	ms_lpd3dDevice->GetViewport(&ms_Viewport);

	m_pStateManager = new CStateManager(ms_lpd3dDevice);

	D3DXCreateMatrixStack(0, &ms_lpd3dMatStack);
	ms_lpd3dMatStack->LoadIdentity();

	ms_ptVS	= CreatePTStreamVertexShader();
	ms_pntVS = CreatePNTStreamVertexShader();
	ms_pnt2VS = CreatePNT2StreamVertexShader();

	D3DXMatrixIdentity(&ms_matIdentity);
	D3DXMatrixIdentity(&ms_matView);
	D3DXMatrixIdentity(&ms_matProj);
	D3DXMatrixIdentity(&ms_matInverseView);
	D3DXMatrixIdentity(&ms_matInverseViewYAxis);
	D3DXMatrixIdentity(&ms_matScreen0);
	D3DXMatrixIdentity(&ms_matScreen1);
	D3DXMatrixIdentity(&ms_matScreen2);

	ms_matScreen0._11 = 1;
	ms_matScreen0._22 = -1;	

	ms_matScreen1._41 = 1;
	ms_matScreen1._42 = 1;

	ms_matScreen2._11 = (float) iHres / 2;
	ms_matScreen2._22 = (float) iVres / 2;
	
	D3DXCreateSphere(ms_lpd3dDevice, 1.0f, 32, 32, &ms_lpSphereMesh, NULL);
	D3DXCreateCylinder(ms_lpd3dDevice, 1.0f, 1.0f, 1.0f, 8, 8, &ms_lpCylinderMesh, NULL);

	ms_lpd3dDevice->Clear(0L, NULL, D3DCLEAR_TARGET | D3DCLEAR_ZBUFFER, 0xff000000, 1.0f, 0);

	if (!__CreateDefaultIndexBufferList())
		return false;

	if (!__CreatePDTVertexBufferList())
		return false;
	
	DWORD dwTexMemSize = GetAvailableTextureMemory();

	if (dwTexMemSize < 64 * 1024 * 1024)
		ms_isLowTextureMemory = true;
	else
		ms_isLowTextureMemory = false;

	if (dwTexMemSize > 100 * 1024 * 1024)
		ms_isHighTextureMemory = true;
	else
		ms_isHighTextureMemory = false;

	if (ms_d3dCaps.TextureAddressCaps & D3DPTADDRESSCAPS_BORDER)
		GRAPHICS_CAPS_CAN_NOT_TEXTURE_ADDRESS_BORDER=false;
	else
		GRAPHICS_CAPS_CAN_NOT_TEXTURE_ADDRESS_BORDER=true;

	//D3DADAPTER_IDENTIFIER9& rkD3DAdapterId=pkD3DAdapterInfo->GetIdentifier();
	if (strnicmp(d3dAdapterId.Driver, "SIS", 3) == 0)
	{
		GRAPHICS_CAPS_CAN_NOT_DRAW_LINE = true;
		GRAPHICS_CAPS_CAN_NOT_DRAW_SHADOW = true;
		GRAPHICS_CAPS_HALF_SIZE_IMAGE = true;
		ms_isLowTextureMemory = true;
	}
	else if (strnicmp(d3dAdapterId.Driver, "3dfx", 4) == 0)
	{
		GRAPHICS_CAPS_CAN_NOT_DRAW_SHADOW = true;
		GRAPHICS_CAPS_HALF_SIZE_IMAGE = true;
		ms_isLowTextureMemory = true;
	}

	// CPU check
	int cpuInfo[4] = { 0 };
	__cpuid(cpuInfo, 1);

	CPU_HAS_SSE2 = cpuInfo[3] & (1 << 26);

	return (iRet);
}

{
	assert(ms_alpd3dDefIB[eDefIB]==NULL);

	auto hr = ms_lpd3dDevice->CreateIndexBuffer(
		sizeof(WORD)*uIdxCount,
		D3DUSAGE_WRITEONLY,
		D3DFMT_INDEX16,
		D3DPOOL_DEFAULT,
		&ms_alpd3dDefIB[eDefIB], nullptr
	);
	if (FAILED(hr))
		return false;

	hr = ms_alpd3dDefIB[eDefIB]->Lock(0, 0, (void**)&dstIndices, 0);

	return true;
}

		{

		case S_OK: return "S_OK";

		case D3DERR_DEVICELOST: return "D3DERR_DEVICELOST";

		case D3DERR_DEVICEHUNG: return "D3DERR_DEVICEHUNG";

		case D3DERR_OUTOFVIDEOMEMORY: return "D3DERR_OUTOFVIDEOMEMORY";

		}

	}

void CGraphicDevice::RecordDeviceResult(HRESULT result, const char* operation)

{

	if (!CheckRenderState())

		return;



	if (result != m_deviceResult)

	{

		if (FAILED(result))

			TraceError("Graphics: %s returned %s (HRESULT=0x%08lX, windowed=%d, backbuffer=%ux%u)",

				operation, DeviceResultName(result), static_cast<unsigned long>(result), ms_d3dPresentParameter.Windowed,

				ms_d3dPresentParameter.BackBufferWidth, ms_d3dPresentParameter.BackBufferHeight);

		else

			Tracenf("Graphics: %s returned %s (0x%08lX)", operation, DeviceResultName(result), static_cast<unsigned long>(result));

	}

	m_deviceResult = result;



	if (result == S_PRESENT_MODE_CHANGED)

		m_resetPending = true;

	else if (result == D3DERR_DEVICELOST)

	{

		m_resetPending = true;

		if (!m_recovering)

		{

			m_recovering = true;

			m_recoveryStartTime = GetTickCount();

		}

	}

	else if (result != S_OK && result != S_PRESENT_OCCLUDED)

	{

		// A hung device can have lost texture contents; a removed device needs

		// full recreation. Neither is repaired by a presentation-only reset.

		m_fatalDeviceError = true;

		TraceError("Graphics: stopping client loop; %s returned 0x%08lX. Restart required.",

			operation, static_cast<unsigned long>(result));

	}

}



bool CGraphicDevice::CheckRenderState()

{

	const auto& status = CGraphicsRestoreStatus::Instance();

	if (!m_fatalDeviceError && status.IsBlocked())

	{

		m_deviceResult = status.Result();

		m_fatalDeviceError = true;

		TraceError("Graphics: stopping client loop after %s (HRESULT=0x%08lX). Restart required.",

			status.Operation(), static_cast<unsigned long>(status.Result()));

	}

	return !m_fatalDeviceError;

}



CGraphicDevice::EFrameStatus CGraphicDevice::PrepareFrame(bool minimized)

{

	if (!CheckRenderState() || !ms_lpd3dDevice)

		return FRAME_FATAL;

	if (m_resetInProgress)

		return FRAME_RETRY;



	const DWORD now = GetTickCount();

	if (m_recovering && now - m_recoveryStartTime >= DeviceRecoveryTimeoutMs)

	{

		TraceError("Graphics: recovery timed out after %lu ms; stopping client loop.", DeviceRecoveryTimeoutMs);

		m_fatalDeviceError = true;

		return FRAME_FATAL;

	}



	// Healthy frames do not poll the driver. Occlusion and transient loss are

	// checked at a bounded rate after presentation/scene/reset reports trouble.

	if (m_deviceResult != S_OK)

	{

		if (now - m_lastDeviceCheckTime < DeviceCheckIntervalMs)

			return m_recovering ? FRAME_RETRY : FRAME_SKIP_RENDER;



		m_lastDeviceCheckTime = now;

		RecordDeviceResult(ms_lpd3dDevice->CheckDeviceState(ms_hWnd), "CheckDeviceState");

		if (m_fatalDeviceError)

			return FRAME_FATAL;

		// DEVICELOST after a failed reset requires another ResetEx attempt;

		// waiting for the obsolete DEVICENOTRESET status would never recover.

		if (m_deviceResult == S_PRESENT_OCCLUDED)

			return m_recovering ? FRAME_RETRY : FRAME_SKIP_RENDER;

	}



	if (minimized)

		return m_recovering ? FRAME_RETRY : FRAME_SKIP_RENDER;



	if (m_resetPending && !__ResetPending())

		return m_fatalDeviceError ? FRAME_FATAL : FRAME_RETRY;



	m_recovering = false;

	return FRAME_READY;

}



bool CGraphicDevice::__ResetPending()

{

	// ResetEx mutates its arguments and can dispatch window messages. Keep the

	// successful settings and any newly queued request separate from its copy.

	const D3DPRESENT_PARAMETERS requested = m_pendingPresentParameters;

	D3DPRESENT_PARAMETERS parameters = requested;

	D3DDISPLAYMODEEX displayMode = m_fullscreenDisplayMode;

	if (parameters.Windowed)

		parameters.BackBufferFormat = D3DFMT_UNKNOWN;

	const bool browserMode = m_pendingBrowserMode;

	if (!m_recovering)

		Tracenf("Graphics: ResetEx requested (windowed=%d, backbuffer=%ux%u)",

			requested.Windowed, requested.BackBufferWidth, requested.BackBufferHeight);

	m_resetPending = false;

	m_resetInProgress = true;

	const HRESULT result = ms_lpd3dDevice->ResetEx(&parameters,

		requested.Windowed ? nullptr : &displayMode);

	m_resetInProgress = false;

	RecordDeviceResult(result, "ResetEx");

	if (FAILED(result))

	{

		m_resetPending = true;

		m_lastDeviceCheckTime = GetTickCount();

		return false;

	}



	// Read the actual settings instead of retaining an old desktop format or

	// the reset API's modified input fields. Release the swap-chain reference

	// immediately so it cannot interfere with a later reset.

	IDirect3DSwapChain9* swapChain = nullptr;

	D3DPRESENT_PARAMETERS actual = {};

	HRESULT settingsResult = ms_lpd3dDevice->GetSwapChain(0, &swapChain);

	if (SUCCEEDED(settingsResult))

	{

		settingsResult = swapChain->GetPresentParameters(&actual);

		swapChain->Release();

	}

	RecordDeviceResult(settingsResult, "Read presentation settings after ResetEx");

	if (FAILED(settingsResult))

		return false;

	ms_d3dPresentParameter = actual;

	g_isBrowserMode = browserMode;



	// Ex preserves resources and render states. Only the main viewport changes;

	// resetting the state cache here would erase the retained transforms.

	D3DVIEWPORT9 viewport = { 0, 0, actual.BackBufferWidth, actual.BackBufferHeight, 0.0f, 1.0f };

	const HRESULT viewportResult = ms_lpd3dDevice->SetViewport(&viewport);

	RecordDeviceResult(viewportResult, "SetViewport after ResetEx");

	if (FAILED(viewportResult))

		return false;

	ms_Viewport = viewport;

	Tracenf("Graphics: ResetEx complete (windowed=%d, backbuffer=%ux%u)",

		actual.Windowed, actual.BackBufferWidth, actual.BackBufferHeight);

	return true;

}

bool CGraphicDevice::__CreateDefaultIndexBuffer(UINT eDefIB, UINT uIdxCount, const WORD* c_awIndices)

{

	assert(ms_alpd3dDefIB[eDefIB]==NULL);



	const auto check = [&](HRESULT result, const char* operation)

	{

		ms_hLastResult = result;

		if (FAILED(result)) TraceError("Graphics: startup index buffer %u %s failed (HRESULT=0x%08lX)", eDefIB, operation, static_cast<unsigned long>(result));

		return SUCCEEDED(result);

	};

	auto hr = ms_lpd3dDevice->CreateIndexBuffer(

		sizeof(WORD)*uIdxCount,

		D3DUSAGE_WRITEONLY,

		D3DFMT_INDEX16,

		D3DPOOL_DEFAULT,

		&ms_alpd3dDefIB[eDefIB], nullptr

	);

	if (!check(hr, "CreateIndexBuffer"))

		return false;

	

	WORD* dstIndices = nullptr;

	hr = ms_alpd3dDefIB[eDefIB]->Lock(0, 0, (void**)&dstIndices, 0);

	if (!check(hr, "Lock"))

		return false;

	if (!dstIndices)

	{

		ms_alpd3dDefIB[eDefIB]->Unlock();

		check(E_FAIL, "null locked data");

		return false;

	}



	memcpy(dstIndices, c_awIndices, sizeof(WORD)*uIdxCount);



	return check(ms_alpd3dDefIB[eDefIB]->Unlock(), "Unlock");

}

						float fNear=MIN_FOG;
						double dbAvePow=double(1000.0f/s_fAveRenderTime);

			}
		}
	}

	int rest = s_uiNextFrameTime - ELTimer_GetMSec();

		s_uiLoad -= rest;	// ½® ½Ã°£Àº ·Îµå¿¡¼­ »«´Ù..

	}	

	++s_dwUpdateFrameCount;

	s_uiLoad += ELTimer_GetMSec() - dwStart;
	//m_Profiler.ProfileByScreen();	
	return true;
}

int CPythonApplication::CheckDeviceState()
{
	CGraphicDevice::EDeviceState e_deviceState = m_grpDevice.GetDeviceState();

	switch (e_deviceState)
	{
		// µð¹ÙÀÌ½º°¡ ¾øÀ¸¸é ÇÁ·Î±×·¥ÀÌ Á¾·á µÇ¾î¾ß ÇÑ´Ù.
	case CGraphicDevice::DEVICESTATE_NULL:
		return DEVICE_STATE_FALSE;

		// DEVICESTATE_BROKENÀÏ ¶§´Â ´ÙÀ½ ·çÇÁ¿¡¼­ º¹±¸ µÉ ¼ö ÀÖµµ·Ï ¸®ÅÏ ÇÑ´Ù.
		// ±×³É ÁøÇàÇÒ °æ¿ì DrawPrimitive °°Àº °ÍÀ» ÇÏ¸é ÇÁ·Î±×·¥ÀÌ ÅÍÁø´Ù.
	case CGraphicDevice::DEVICESTATE_BROKEN:
		return DEVICE_STATE_SKIP;

	case CGraphicDevice::DEVICESTATE_NEEDS_RESET:
		if (!m_grpDevice.Reset())
			return DEVICE_STATE_SKIP;

		break;
	}

	return DEVICE_STATE_OK;
}

bool CPythonApplication::CreateDevice(int width, int height, int Windowed, int bit, int frequency)
{
	int iRet;

	m_grpDevice.InitBackBufferCount(2);
	iRet = m_grpDevice.Create(GetWindowHandle(), width, height, Windowed ? true : false, bit, frequency);

	switch (iRet)
	{
		case CGraphicDevice::CREATE_OK:
			return true;

		case CGraphicDevice::CREATE_REFRESHRATE:
			return true;

		case CGraphicDevice::CREATE_ENUM:
		case CGraphicDevice::CREATE_DETECT:
			SET_EXCEPTION(CREATE_NO_APPROPRIATE_DEVICE);
			TraceError("CreateDevice: Enum & Detect failed");
			return false;

		case CGraphicDevice::CREATE_NO_DIRECTX:
			SET_EXCEPTION(CREATE_NO_DIRECTX);
			TraceError("CreateDevice: DirectX 8.1 or greater required to run game");
			return false;

		case CGraphicDevice::CREATE_DEVICE:
			SET_EXCEPTION(CREATE_DEVICE);
			TraceError("CreateDevice: GraphicDevice create failed");
			return false;

		case CGraphicDevice::CREATE_FORMAT:
			SET_EXCEPTION(CREATE_FORMAT);
			TraceError("CreateDevice: Change the screen format");
			return false;

		case CGraphicDevice::CREATE_GET_DEVICE_CAPS:
			PyErr_SetString(PyExc_RuntimeError, "GetDevCaps failed");
			TraceError("CreateDevice: GetDevCaps failed");
			return false;

		case CGraphicDevice::CREATE_GET_DEVICE_CAPS2:
			PyErr_SetString(PyExc_RuntimeError, "GetDevCaps2 failed");
			TraceError("CreateDevice: GetDevCaps2 failed");
			return false;

		default:
			if (iRet & CGraphicDevice::CREATE_OK)
			{
				if (iRet & CGraphicDevice::CREATE_NO_TNL)
				{
					CGrannyLODController::SetMinLODMode(true);
				}
				return true;
			}

			SET_EXCEPTION(UNKNOWN_ERROR);
			TraceError("CreateDevice: Unknown Error!");
			return false;
	}
}

void CPythonApplication::UpdateMainWindowPosition()
{
	POINT finalPoint{};
	if (GetCursorPos(&finalPoint))
	{
		LONG xDiff = finalPoint.x - m_InitialMouseMovingPoint.x;
		LONG yDiff = finalPoint.y - m_InitialMouseMovingPoint.y;

		RECT r{};
		GetWindowRect(&r);

		SetPosition(r.left + xDiff, r.top + yDiff);
		m_InitialMouseMovingPoint = finalPoint;
	}
}

void CPythonApplication::Loop()
{	
	while (1)
	{	
		if (IsUserMovingMainWindow())
			UpdateMainWindowPosition();
		
		if (IsMessage())
		{
			if (!MessageProcess())
				break;
		}
		else
		{
			if (!Process())
				break;

			m_dwLastIdleTime=ELTimer_GetMSec();
		}
	}
}

			return true;



		case CGraphicDevice::CREATE_ENUM:

		case CGraphicDevice::CREATE_DETECT:

			SET_EXCEPTION(CREATE_NO_APPROPRIATE_DEVICE);

		case CGraphicDevice::CREATE_DEVICE:

		case CGraphicDevice::CREATE_FORMAT:

			SET_EXCEPTION(CREATE_FORMAT);

			TraceError("CreateDevice: Change the screen format");

			return false;



		case CGraphicDevice::CREATE_GET_DEVICE_CAPS:

			PyErr_SetString(PyExc_RuntimeError, "GetDevCaps failed");

			TraceError("CreateDevice: GetDevCaps failed");

			return false;



		case CGraphicDevice::CREATE_GET_DEVICE_CAPS2:

			PyErr_SetString(PyExc_RuntimeError, "GetDevCaps2 failed");

			TraceError("CreateDevice: GetDevCaps2 failed");

			return false;

					CGrannyLODController::SetMinLODMode(true);

						UINT uWidth=rcWnd.right-rcWnd.left; 

						UINT uHeight=rcWnd.bottom-rcWnd.top;

						m_grpDevice.RequestBackBufferResize(uWidth, uHeight);

					}

					break;

			}



			if (wParam==SIZE_MINIMIZED)

				m_isMinimizedWnd=true;

			else

				m_isMinimizedWnd=false;



			OnSizeChange(short(LOWORD(lParam)), short(HIWORD(lParam)));



			break;



		case WM_EXITSIZEMOVE:    

			{

				RECT rcWnd; 

				GetClientRect(&rcWnd); 

				

				UINT uWidth=rcWnd.right-rcWnd.left; 

				UINT uHeight=rcWnd.bottom-rcWnd.top;

				m_grpDevice.RequestBackBufferResize(uWidth, uHeight);

				OnSizeChange(uWidth, uHeight);

			}

			break; 

		case WM_NCLBUTTONDOWN:

			{

				switch (wParam)

				{

				case HTMAXBUTTON:

				case HTSYSMENU:

					return 0;

				case HTMINBUTTON:

					ShowWindow(hWnd, SW_MINIMIZE);

					return 0;

				case HTCLOSE:

					RunPressExitKey();

					return 0;

				case HTCAPTION:

					if (!IsUserMovingMainWindow())

						SetUserMovingMainWindow(true);

		

					return 0;

				}

		

				break;

			}

			

		case WM_NCLBUTTONUP:

			{

				if (IsUserMovingMainWindow())

					SetUserMovingMainWindow(false);

				

				break;

			}

		

		case WM_NCRBUTTONDOWN:

		case WM_NCRBUTTONUP:

		case WM_CONTEXTMENU:

			return 0;

		case WM_SYSCOMMAND:

			if (wParam == SC_KEYMENU)

				return 0;

			break;

		case WM_SYSKEYDOWN:

			switch (LOWORD(wParam))

			{

				case VK_F10:

					break;

			}

			break;



		case WM_SYSKEYUP:

			switch(LOWORD(wParam))

			{

				case 18:

					return FALSE;

					break;

				case VK_F10:

					break;

			}

			break;



		case WM_SETCURSOR:

			if (IsActive())

			{

				if (m_bCursorVisible && CURSOR_MODE_HARDWARE == m_iCursorMode)

				{

					SetCursor((HCURSOR) m_hCurrentCursor);

					return 0;

				}

				else

				{

					SetCursor(NULL);

					return 0;

				}

			}

			break;



		case WM_CLOSE:

#ifdef _DEBUG

			PostQuitMessage(0);

#else	

			RunPressExitKey();

#endif

			return 0;



		case WM_DESTROY:

			return 0;

		default:

			//Tracenf("%x msg %x", timeGetTime(), uiMsg);

			break;

	}	



	return CMSApplication::WindowProcedure(hWnd, uiMsg, wParam, lParam);

}
