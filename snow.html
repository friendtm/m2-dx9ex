		std::vector<CSnowParticle*>::iterator itor = m_kVct_pkParticleSnow.begin();

		for (; i < dwParticleCount && itor != m_kVct_pkParticleSnow.end(); ++i, ++itor)

		{

			CSnowParticle * pSnow = *itor;

			pSnow->SetCameraVertex(c_rv3Up, c_rv3Cross);

			pSnow->GetVerticies(pv3Verticies[i*4+0],

								pv3Verticies[i*4+1],

								pv3Verticies[i*4+2],

								pv3Verticies[i*4+3]);

		}

		if (FAILED(m_pVB->Unlock())) return false;

	}

	else return false; // Never draw stale geometry after a failed upload.



	CStateManager::ScopedState drawingState(STATEMANAGER);

	if (!drawingState.IsValid()) return false;

	CRenderTargetBinding blurTarget; // Restores before drawingState on every exit.

	D3DVIEWPORT9 originalViewport = {};

	if (m_bBlurEnable)

	{

		if (!m_lpSnowTexture || !m_lpSnowRenderTargetSurface || !m_lpSnowDepthSurface) return false;

		if (FAILED(ms_lpd3dDevice->GetViewport(&originalViewport))) return false;

		const D3DVIEWPORT9 viewport = {0, 0, m_wBlurTextureSize, m_wBlurTextureSize, 0.0f, 1.0f};

		if (FAILED(blurTarget.Bind(ms_lpd3dDevice, m_lpSnowRenderTargetSurface, m_lpSnowDepthSurface, viewport))) return false;

		if (FAILED(ms_lpd3dDevice->Clear(0, nullptr, D3DCLEAR_TARGET | D3DCLEAR_ZBUFFER, 0, 1.0f, 0))) return false;

	}



	if (FAILED(STATEMANAGER.SetRenderState(D3DRS_ZWRITEENABLE, FALSE))) return false;

	if (FAILED(STATEMANAGER.SetRenderState(D3DRS_ALPHABLENDENABLE, TRUE))) return false;

	if (FAILED(STATEMANAGER.SetRenderState(D3DRS_CULLMODE, D3DCULL_NONE))) return false;

	if (FAILED(STATEMANAGER.SetRenderState(D3DRS_SRCBLEND,  D3DBLEND_SRCALPHA))) return false;

	if (FAILED(STATEMANAGER.SetRenderState(D3DRS_DESTBLEND, D3DBLEND_INVSRCALPHA))) return false;

	if (FAILED(STATEMANAGER.SetTextureStageState(0, D3DTSS_COLORARG1, D3DTA_TEXTURE))) return false;

	if (FAILED(STATEMANAGER.SetTextureStageState(0, D3DTSS_COLOROP, D3DTOP_SELECTARG1))) return false;

	if (FAILED(STATEMANAGER.SetTextureStageState(0, D3DTSS_ALPHAARG1, D3DTA_TEXTURE))) return false;

	if (FAILED(STATEMANAGER.SetTextureStageState(0, D3DTSS_ALPHAOP, D3DTOP_SELECTARG1))) return false;

	if (FAILED(STATEMANAGER.SetTexture(1, NULL))) return false;

	if (FAILED(STATEMANAGER.SetTextureStageState(1, D3DTSS_COLOROP, D3DTOP_DISABLE))) return false;

	if (FAILED(STATEMANAGER.SetTextureStageState(1, D3DTSS_ALPHAOP, D3DTOP_DISABLE))) return false;



	if (FAILED(STATEMANAGER.SetTexture(0, m_pImageInstance->GetGraphicImagePointer()->GetTextureReference().GetD3DTexture()))) return false;

	if (FAILED(STATEMANAGER.SetIndices(m_pIB, 0))) return false;

	if (FAILED(STATEMANAGER.SetStreamSource(0, m_pVB, sizeof(SParticleVertex)))) return false;

	if (FAILED(STATEMANAGER.SetFVF(D3DFVF_XYZ | D3DFVF_TEX1))) return false;

	if (FAILED(STATEMANAGER.DrawIndexedPrimitive(D3DPT_TRIANGLELIST, 0, dwParticleCount*4, 0, dwParticleCount*2))) return false;

	if (m_bBlurEnable)

	{

		if (FAILED(blurTarget.Restore())) return false;

		if (FAILED(STATEMANAGER.SetTexture(0, m_lpSnowTexture))) return false;

		const float x = static_cast<float>(originalViewport.X), y = static_cast<float>(originalViewport.Y);

		const float right = x + originalViewport.Width, bottom = y + originalViewport.Height;

		BlurVertex vertices[4] = {

			BlurVertex(D3DXVECTOR3(x,y,0),1,0xFFFFFF,0,0),

			BlurVertex(D3DXVECTOR3(right,y,0),1,0xFFFFFF,1,0),

			BlurVertex(D3DXVECTOR3(x,bottom,0),1,0xFFFFFF,0,1),

			BlurVertex(D3DXVECTOR3(right,bottom,0),1,0xFFFFFF,1,1)};

		if (FAILED(STATEMANAGER.SetFVF(D3DFVF_XYZRHW | D3DFVF_DIFFUSE | D3DFVF_TEX1))) return false;

		if (FAILED(STATEMANAGER.DrawPrimitiveUP(D3DPT_TRIANGLESTRIP, 2, vertices, sizeof(BlurVertex)))) return false;

	}

	return SUCCEEDED(drawingState.Restore());

}



bool CSnowEnvironment::__CreateBlurTexture()

{

	if (!m_bBlurEnable)

		return true;



	if (FAILED(ms_lpd3dDevice->CreateTexture(m_wBlurTextureSize, m_wBlurTextureSize, 1, D3DUSAGE_RENDERTARGET, D3DFMT_X8R8G8B8, D3DPOOL_DEFAULT, &m_lpSnowTexture, nullptr)))

		return false;

	if (FAILED(m_lpSnowTexture->GetSurfaceLevel(0, &m_lpSnowRenderTargetSurface)))

		return false;

	if (FAILED(ms_lpd3dDevice->CreateDepthStencilSurface(m_wBlurTextureSize, m_wBlurTextureSize, D3DFMT_D16, D3DMULTISAMPLE_NONE, 0, TRUE, &m_lpSnowDepthSurface, nullptr)))

		return false;



	return true;

}



bool CSnowEnvironment::__CreateGeometry()

{

	if (!ms_lpd3dDevice || m_pVB || m_pIB || !m_dwParticleMaxNum || m_dwParticleMaxNum > 16384) return false;

	if (FAILED(ms_lpd3dDevice->CreateVertexBuffer(sizeof(SParticleVertex) * m_dwParticleMaxNum * 4,

		D3DUSAGE_DYNAMIC | D3DUSAGE_WRITEONLY,

		D3DFVF_XYZ | D3DFVF_TEX1,

		D3DPOOL_DEFAULT,

		&m_pVB, nullptr)))

		return false;



	if (FAILED(ms_lpd3dDevice->CreateIndexBuffer(sizeof(WORD) * m_dwParticleMaxNum * 6,

		D3DUSAGE_DYNAMIC,

		D3DFMT_INDEX16,

		D3DPOOL_DEFAULT,

		&m_pIB, nullptr)))

		return false;



	WORD* dstIndices = nullptr;

	if (!GeometryBufferUpload::Lock(m_pIB, sizeof(WORD) * m_dwParticleMaxNum * 6, (void**)&dstIndices))

		return false;



	const WORD c_awFillRectIndices[6] = { 0, 2, 1, 2, 3, 1, };

	for (int i = 0; i < m_dwParticleMaxNum; ++i)

	{

		for (int j = 0; j < 6; ++j)

		{

			dstIndices[i*6 + j] = i*4 + c_awFillRectIndices[j];

		}

	}



	return SUCCEEDED(m_pIB->Unlock());

}



bool CSnowEnvironment::Create()

{

	Destroy();

	const auto cleanup = [this]() { Destroy(); };

	GraphicsStartup::Rollback<decltype(cleanup)> rollback(cleanup);

	if (!ms_lpd3dDevice) return false;



	if (!__CreateBlurTexture())

		return false;



	if (!__CreateGeometry())

		return false;



	CResource* resource = CResourceManager::Instance().GetResourcePointer("d:/ymir work/special/snow.dds");

	if (!resource || !resource->IsType(CGraphicImage::Type())) return false;

	CGraphicImage* pImage = static_cast<CGraphicImage*>(resource);

	m_pImageInstance = CGraphicImageInstance::New();

	if (!m_pImageInstance) return false;

	m_pImageInstance->SetImagePointer(pImage);

	if (m_pImageInstance->IsEmpty() || !pImage->GetTextureReference().GetD3DTexture()) return false;

	rollback.Complete();

	return true;

}



void CSnowEnvironment::Destroy()

}
