	class ControlledUploadBackend

	{

	public:

		enum class Fault { None, Staging, Lock, NullBits, ZeroPitch, NegativePitch,

			ShortPitch, Unlock, Destination, Update };

		static constexpr size_t Guard = 16;

		static constexpr uint8_t Canary = 0xCD;

		struct Texture

		{

			UINT width = 0, height = 0;

			int pitch = 0;

			D3DPOOL pool = D3DPOOL_SYSTEMMEM;

			bool locked = false;

			std::vector<uint8_t> bytes;

		};

		Fault fault = Fault::None;

		int padding = 16;

		unsigned creates = 0, locks = 0, unlocks = 0, updates = 0, releases = 0;

		bool cleanupError = false;



		HRESULT Create(UINT width, UINT height, D3DPOOL pool, Texture** output)

		{

			++creates;

			*output = nullptr;

			if (pool == D3DPOOL_SYSTEMMEM && fault == Fault::Staging) return E_OUTOFMEMORY;

			if (pool == D3DPOOL_DEFAULT && fault == Fault::Destination) return D3DERR_OUTOFVIDEOMEMORY;

			auto texture = std::make_unique<Texture>();

			texture->width = width;

			texture->height = height;

			texture->pitch = static_cast<int>(width * 4) + padding;

			texture->pool = pool;

			texture->bytes.assign(Guard * 2 + static_cast<size_t>(texture->pitch) * height, Canary);

			m_live.push_back(std::move(texture));

			*output = m_live.back().get();

			return S_OK;

		}



		HRESULT Lock(Texture* texture, D3DLOCKED_RECT* locked)

		{

			++locks;

			Require(texture->pool == D3DPOOL_SYSTEMMEM && !texture->locked, "Invalid controlled lock sequence");

			if (fault == Fault::Lock) return D3DERR_INVALIDCALL;

			texture->locked = true;

			locked->pBits = texture->bytes.data() + Guard;

			locked->Pitch = texture->pitch;

			if (fault == Fault::NullBits) locked->pBits = nullptr;

			if (fault == Fault::ZeroPitch) locked->Pitch = 0;

			if (fault == Fault::NegativePitch) locked->Pitch = -1;

			if (fault == Fault::ShortPitch) locked->Pitch = static_cast<int>(texture->width * 4) - 1;

			return S_OK;

		}



		HRESULT Unlock(Texture* texture)

		{

			++unlocks;

			Require(texture->locked, "Unlock without a successful lock");

			if (fault == Fault::Unlock) return E_FAIL;

			texture->locked = false;

			return S_OK;

		}



		HRESULT Update(Texture* source, Texture* destination)

		{

			++updates;

			Require(!source->locked && source->pool == D3DPOOL_SYSTEMMEM &&

				destination->pool == D3DPOOL_DEFAULT && source->width == destination->width &&

				source->height == destination->height, "Invalid controlled update sequence");

			if (fault == Fault::Update)

			{

				// A failed copy may have partially touched its destination.

				destination->bytes[Guard] = 0;

				return D3DERR_DEVICELOST;

			}

			for (UINT row = 0; row < source->height; ++row)

				memcpy(destination->bytes.data() + Guard + static_cast<size_t>(row) * destination->pitch,

					source->bytes.data() + Guard + static_cast<size_t>(row) * source->pitch, source->width * 4);

			return S_OK;

		}



		bool GuardsIntact(const Texture& texture) const noexcept

		{

			for (size_t index = 0; index < texture.bytes.size(); ++index)

			{

				const bool outside = index < Guard || index >= texture.bytes.size() - Guard;

				const bool paddingByte = !outside && (index - Guard) % texture.pitch >= texture.width * 4;

				if ((outside || paddingByte) && texture.bytes[index] != Canary) return false;

			}

			return true;

		}



		void Release(Texture* texture) noexcept

		{

			for (auto item = m_live.begin(); item != m_live.end(); ++item)

				if (item->get() == texture)

				{

					// A failed Unlock is followed by release, not another copy attempt.

					cleanupError |= !GuardsIntact(*texture) || (texture->locked && fault != Fault::Unlock);

					if (fault == Fault::NullBits || fault == Fault::ZeroPitch ||

						fault == Fault::NegativePitch || fault == Fault::ShortPitch)

						for (const auto byte : texture->bytes)

							cleanupError |= byte != Canary; // Invalid locks must cause no writes at all.

					++releases;

					m_live.erase(item);

					return;

				}

			cleanupError = true; // Double release or a foreign handle.

		}



		size_t LiveCount() const { return m_live.size(); }

		void ResetCounters()

		{

			Require(m_live.empty() && !cleanupError, "Resources remained before retry");

			creates = locks = unlocks = updates = releases = 0;

		}

	private:

		// Own even leaked test handles so a failed assertion cannot leak the harness.

		// LiveCount is checked BEFORE this fallback cleanup occurs.

		std::vector<std::unique_ptr<Texture>> m_live;

	};

	int VerifyPixels(IDirect3DDevice9Ex* device, CGraphicImageTexture& texture,

		const TDecodedImageData& image, Report& report)

	{

		Require(!texture.IsEmpty() && texture.GetD3DTexture(), "Upload returned no usable texture");

		D3DSURFACE_DESC description = {};

		Check(texture.GetD3DTexture()->GetLevelDesc(0, &description), "GetLevelDesc");

		Require(description.Width == static_cast<UINT>(image.width) &&

			description.Height == static_cast<UINT>(image.height) &&

			description.Format == D3DFMT_A8R8G8B8 && description.Pool == D3DPOOL_DEFAULT &&

			texture.GetD3DTexture()->GetLevelCount() == 1 &&

			texture.GetWidth() == image.width && texture.GetHeight() == image.height,

			"Uploaded texture does not match its dimensions/format/pool/level contract");

		ComPtr<IDirect3DSurface9> target;

		ComPtr<IDirect3DSurface9> readback;

		Check(device->CreateRenderTarget(image.width, image.height, D3DFMT_A8R8G8B8,

			D3DMULTISAMPLE_NONE, 0, FALSE, target.GetAddressOf(), nullptr), "CreateRenderTarget");

		Check(device->CreateOffscreenPlainSurface(image.width, image.height, D3DFMT_A8R8G8B8,

			D3DPOOL_SYSTEMMEM, readback.GetAddressOf(), nullptr), "Create readback surface");

		DrawBinding binding(device);

		Check(device->SetRenderTarget(0, target.Get()), "SetRenderTarget");

		const D3DVIEWPORT9 viewport = {0, 0, static_cast<DWORD>(image.width),

			static_cast<DWORD>(image.height), 0.0f, 1.0f};

		Check(device->SetViewport(&viewport), "SetViewport");

		Check(device->Clear(0, nullptr, D3DCLEAR_TARGET, 0x2468ACE0, 1.0f, 0), "Clear");

		Check(device->SetTexture(0, texture.GetD3DTexture()), "SetTexture");

		struct Vertex { float x, y, z, rhw, u, v; };

		// D3D9 pixel centres lie on integers: offset the quad edges by half a pixel.

		const float right = static_cast<float>(image.width) - 0.5f;

		const float bottom = static_cast<float>(image.height) - 0.5f;

		const Vertex vertices[] = {

			{-0.5f, -0.5f, 0, 1, 0, 0}, {right, -0.5f, 0, 1, 1, 0},

			{-0.5f, bottom, 0, 1, 0, 1}, {right, bottom, 0, 1, 1, 1}

		};

		Check(device->BeginScene(), "BeginScene");

		const HRESULT draw = device->DrawPrimitiveUP(D3DPT_TRIANGLESTRIP, 2, vertices, sizeof(Vertex));

		const HRESULT end = device->EndScene(); // Always end a successfully begun scene.

		Check(draw, "DrawPrimitiveUP");

		Check(end, "EndScene");

		binding.Clear();

		Check(device->GetRenderTargetData(target.Get(), readback.Get()), "GetRenderTargetData");

		D3DLOCKED_RECT locked = {};

		Check(readback->LockRect(&locked, nullptr, D3DLOCK_READONLY), "Lock readback");

		bool matches = locked.pBits && locked.Pitch > 0 &&

			static_cast<size_t>(locked.Pitch) >= static_cast<size_t>(image.width) * 4;

		for (int y = 0; matches && y < image.height; ++y)

			for (int x = 0; matches && x < image.width; ++x)

			{

				const uint8_t* expected = image.pixels.data() + (static_cast<size_t>(y) * image.width + x) * 4;

				const uint8_t* actual = static_cast<const uint8_t*>(locked.pBits) +

					static_cast<size_t>(y) * locked.Pitch + x * 4;

				matches = actual[0] == expected[2] && actual[1] == expected[1] &&

					actual[2] == expected[0] && actual[3] == expected[3];

				if (!matches)

					report.Write("Mismatch at (%d,%d): expected RGBA=%u,%u,%u,%u; actual RGBA=%u,%u,%u,%u",

						x, y, expected[0], expected[1], expected[2], expected[3],

						actual[2], actual[1], actual[0], actual[3]);

			}

		Check(readback->UnlockRect(), "Unlock readback");

		Require(matches, "GPU readback mismatch or invalid readback pitch");

		return locked.Pitch;

	}

	bool SupportsSize(const D3DCAPS9& caps, int width, int height)

	{

		if (static_cast<DWORD>(width) > caps.MaxTextureWidth || static_cast<DWORD>(height) > caps.MaxTextureHeight)

			return false;

		if ((caps.TextureCaps & D3DPTEXTURECAPS_SQUAREONLY) && width != height)

			return false;

		if (caps.MaxTextureAspectRatio &&

			(static_cast<uint64_t>(width) > static_cast<uint64_t>(height) * caps.MaxTextureAspectRatio ||

			 static_cast<uint64_t>(height) > static_cast<uint64_t>(width) * caps.MaxTextureAspectRatio))

			return false;

		const bool nonPowerOfTwo = (width & (width - 1)) || (height & (height - 1));

		return !nonPowerOfTwo || !(caps.TextureCaps & D3DPTEXTURECAPS_POW2) ||

			(caps.TextureCaps & D3DPTEXTURECAPS_NONPOW2CONDITIONAL);

	}

	void RunControlledRestoreCases(Report& report)

	{

		report.Write("RUN controlled restoration failures (simulated HRESULTs; no driver faults)");

		struct Backend

		{

			int failStep;

			HRESULT failure;

			unsigned visited = 0;

			HRESULT Step(unsigned step)

			{

				Require(visited == step, "Restoration order changed");

				++visited;

				return failStep == static_cast<int>(step) ? failure : S_OK;

			}

			HRESULT DetachDepth() { return Step(0); }

			HRESULT RestoreTarget() { return Step(1); }

			HRESULT RestoreDepth() { return Step(2); }

			HRESULT RestoreViewport() { return Step(3); }

		};

		const HRESULT failures[] = {E_FAIL, D3DERR_INVALIDCALL, D3DERR_DEVICELOST,

			D3DERR_DEVICEHUNG, D3DERR_DEVICEREMOVED, E_OUTOFMEMORY};

		for (const HRESULT failure : failures)

		{

			for (int step = 0; step < 4; ++step)

			{

				CGraphicsRestoreStatus status;

				Backend backend{step, failure};

				Require(GraphicsRestoreDetail::RestoreTargets(backend, status) == failure,

					"Restoration failure HRESULT lost");

				Require(backend.visited == 4 && status.Result() == failure, "Cleanup stopped early or failure not latched");

				const char* firstOperation = status.Operation();

				Backend successfulCleanup{-1, S_OK};

				Check(GraphicsRestoreDetail::RestoreTargets(successfulCleanup, status), "Later successful cleanup");

				int commits = 0;

				Check(GraphicsRestoreDetail::RestoreDrawingState([]() { return S_OK; }, [&]() { ++commits; }, status),

					"Successful state block after failed target restoration");

				status.Record(E_ABORT, "Later failure");

				Require(commits == 1 && status.Result() == failure && status.Operation() == firstOperation,

					"Later cleanup overwrote first restoration failure");

			}

			CGraphicsRestoreStatus status;

			int commits = 0;

			Require(GraphicsRestoreDetail::RestoreDrawingState([&]() { return failure; },

				[&]() { ++commits; }, status) == failure && status.Result() == failure && commits == 0,

				"Failed state block published saved cache");



			// No HAL device exists here. Exercise the actual owner policy without

			// asking a driver to reset, hang or lose its device.

			CGraphicDevice owner;

			Require(owner.CheckRenderState(), "Fresh owner is unexpectedly blocked");

			CGraphicsRestoreStatus::Instance().Record(failure, "Controlled owner restoration failure");

			Require(owner.HasFatalDeviceError() && owner.AreDeviceCallbacksBlocked(), "Owner callbacks not blocked immediately");

			owner.RecordDeviceResult(S_OK, "Controlled successful EndScene");

			Require(!owner.CheckRenderState() && owner.PrepareFrame(false) == CGraphicDevice::FRAME_FATAL &&

				owner.PrepareFrame(true) == CGraphicDevice::FRAME_FATAL &&

				CGraphicsRestoreStatus::Instance().Result() == failure, "Owner resumed after restoration failure");

			report.Write("PASS controlled restoration HRESULT=0x%08lX: all four target steps, failed Apply/cache, sticky failure, fatal owner",

				static_cast<unsigned long>(failure));

		}

		Require(!CGraphicsRestoreStatus::Instance().IsBlocked(), "Device teardown did not reset restoration status");

		CGraphicsRestoreStatus clean;

		Backend healthy{-1, S_OK};

		Check(GraphicsRestoreDetail::RestoreTargets(healthy, clean), "Healthy restoration");

		Require(!clean.IsBlocked() && healthy.visited == 4, "Healthy restoration poisoned drawing");

		report.Write("Restoration coverage: 24 target-step failures, 6 Apply/cache failures, 6 frame-owner decisions; controlled backend");

	}

	bool RunShaderConstantCases(IDirect3DDevice9Ex* device, const D3DCAPS9& caps, Report& report)

	{

		if (caps.VertexShaderVersion < D3DVS_VERSION(1, 1) || caps.PixelShaderVersion < D3DPS_VERSION(1, 1))

		{

			report.Write("SKIP shader constants: vertex/pixel shader 1.1 support required");

			return false;

		}

		report.Write("RUN shader constant routing: device readback, no shader compilation or execution");

		// Eight registers fit the oldest supported pixel-shader bank. Distinct,

		// exactly representable sentinels detect wrong-stage and wrong-range writes.

		float vertex[8][4], pixel[8][4];

		for (unsigned reg = 0; reg < 8; ++reg)

			for (unsigned component = 0; component < 4; ++component)

			{

				vertex[reg][component] = 0.125f + (reg * 4 + component) / 64.0f;

				pixel[reg][component] = -vertex[reg][component];

			}

		const auto seed = [&]()

		{

			Check(device->SetVertexShaderConstantF(0, &vertex[0][0], 8), "Seed vertex constants");

			Check(device->SetPixelShaderConstantF(0, &pixel[0][0], 8), "Seed pixel constants");

		};

		const auto verify = [&](const float* expectedVertex, const float* expectedPixel)

		{

			float actualVertex[8][4], actualPixel[8][4];

			Check(device->GetVertexShaderConstantF(0, &actualVertex[0][0], 8), "Read vertex constants");

			Check(device->GetPixelShaderConstantF(0, &actualPixel[0][0], 8), "Read pixel constants");

			Require(memcmp(actualVertex, expectedVertex, sizeof(actualVertex)) == 0, "Vertex constant bank mismatch");

			Require(memcmp(actualPixel, expectedPixel, sizeof(actualPixel)) == 0, "Pixel constant bank mismatch");

		};

		seed();

		CStateManager manager(device);

		Check(manager.GetDefaultStateResult(), "Initialize drawing defaults");

		const float zero[8][4] = {};

		verify(&zero[0][0], &zero[0][0]);

		report.Write("PASS shader defaults: constructor clears both seeded constant banks");

		const float replacement[2][4] = {{0.75f, 0.5f, 0.25f, 1.0f}, {-0.75f, -0.5f, -0.25f, -1.0f}};

		const struct { UINT start, count; } ranges[] = {{0, 1}, {3, 2}, {7, 1}};

		for (const auto& range : ranges)

		{

			seed();

			float expectedPixel[8][4];

			memcpy(expectedPixel, pixel, sizeof(pixel));

			memcpy(expectedPixel[range.start], replacement, range.count * sizeof(replacement[0]));

			manager.SetPixelShaderConstant(range.start, replacement, range.count);

			verify(&vertex[0][0], &expectedPixel[0][0]);

			seed();

			float expectedVertex[8][4];

			memcpy(expectedVertex, vertex, sizeof(vertex));

			memcpy(expectedVertex[range.start], replacement, range.count * sizeof(replacement[0]));

			manager.SetVertexShaderConstant(range.start, replacement, range.count);

			verify(&expectedVertex[0][0], &pixel[0][0]);

			report.Write("PASS shader routing: start=%u count=%u; opposite bank and neighboring registers preserved", range.start, range.count);

		}

		seed();

		{

			CStateManager::ScopedState state(manager);

			Require(state.IsValid(), "Capture shader constants");

			manager.SetPixelShaderConstant(3, replacement, 2);

			manager.SetVertexShaderConstant(3, replacement, 2);

			Check(state.Restore(), "Restore shader constants");

		}

		verify(&vertex[0][0], &pixel[0][0]);

		report.Write("PASS shader constant scope: both banks restored");

		Check(manager.SetDefaultState(), "Reinitialize drawing defaults");

		verify(&zero[0][0], &zero[0][0]);

		report.Write("PASS shader defaults: repeated initialization clears both seeded banks");

		report.Write("Shader coverage: 6 stage/range writes, constructor/repeated defaults, scoped restoration; real device state readback");

		return true;

	}



	void RunSamplerCacheCases(IDirect3DDevice9Ex* device, Report& report)

	{

		{

				cached == D3DTEXF_POINT && attempts == 2, "Repeated failed sampler write was suppressed");

			Require(attempts == 3, "Redundant successful sampler write not suppressed");

		}

				verify(stage, type, D3DTEXF_POINT);

			}

			Check(manager.SetSamplerState(stage, D3DSAMP_MIPFILTER, D3DTEXF_NONE), "Disable mipmapping legally");

			verify(stage, D3DSAMP_MIPFILTER, D3DTEXF_NONE);

		}

		Require(manager.SetSamplerState(STATEMANAGER_MAX_STAGES, D3DSAMP_MINFILTER, D3DTEXF_POINT) == D3DERR_INVALIDCALL &&

			manager.SetSamplerState(0, static_cast<D3DSAMPLERSTATETYPE>(0), 0) == D3DERR_INVALIDCALL &&

			manager.SetSamplerState(0, static_cast<D3DSAMPLERSTATETYPE>(0x7fffffff), 0) == D3DERR_INVALIDCALL,

			"Out-of-range sampler cache slot accepted");

		// Call the actual interface setup, not a duplicate of its sampler calls.

		CPythonGraphic graphic;

		Check(manager.SetSamplerState(0, D3DSAMP_MINFILTER, D3DTEXF_LINEAR), "Seed UI min filter");

		Check(manager.SetSamplerState(0, D3DSAMP_MAGFILTER, D3DTEXF_LINEAR), "Seed UI mag filter");

		Check(manager.SetSamplerState(0, D3DSAMP_MIPFILTER, D3DTEXF_LINEAR), "Seed UI mip filter");

		{

			CPreviewRenderState scope(manager);

			Require(scope.IsValid(), "Capture state before UI test");

			graphic.SetInterfaceRenderState();

			verify(0, D3DSAMP_MINFILTER, D3DTEXF_POINT);

			verify(0, D3DSAMP_MAGFILTER, D3DTEXF_POINT);

			verify(0, D3DSAMP_MIPFILTER, D3DTEXF_NONE);

		}

		verify(0, D3DSAMP_MINFILTER, D3DTEXF_LINEAR);

		verify(0, D3DSAMP_MAGFILTER, D3DTEXF_LINEAR);

		verify(0, D3DSAMP_MIPFILTER, D3DTEXF_LINEAR);

		Require(!CGraphicsRestoreStatus::Instance().IsBlocked(), "Healthy sampler restoration blocked rendering");

		report.Write("PASS sampler device/cache: 4 min/mag rejection/reuse and nested-restore cases, mip NONE, slot bounds, production UI setup and scoped restoration");

	}

	void RunGeometryBufferCases(IDirect3DDevice9Ex* device, Report& report)

	{

		struct Buffer

		{

			BYTE data[16] = {};

			HRESULT lockResult = S_OK, unlockResult = S_OK;

			bool nullData = false;

			unsigned locks = 0, unlocks = 0;

			HRESULT Lock(UINT, UINT, void** output, DWORD flags)

			{

				++locks;

				Require(flags == 0, "Geometry copy requested a read-only/discard lock");

				*output = nullData ? nullptr : data;

				return lockResult;

			}

			HRESULT Unlock() { ++unlocks; return unlockResult; }

		};

		const DWORD payload[4] = {1, 2, 3, 4};

		for (int fault = 0; fault < 3; ++fault)

		{

			Buffer buffer;

			if (fault == 0) buffer.lockResult = D3DERR_INVALIDCALL;

			if (fault == 1) buffer.nullData = true;

			if (fault == 2) buffer.unlockResult = E_FAIL;

			Require(!GeometryBufferUpload::Copy(&buffer, 16, 16, payload), "Failed geometry upload reported success");

			Require(buffer.locks == 1 && buffer.unlocks == (fault == 0 ? 0u : 1u), "Geometry lock cleanup mismatch");

			buffer.lockResult = buffer.unlockResult = S_OK; buffer.nullData = false;

			Require(GeometryBufferUpload::Copy(&buffer, 16, 16, payload) && memcmp(buffer.data, payload, 16) == 0, "Geometry retry/copy mismatch");

		}

		Buffer buffer;

		for (int bytes : {-1, 0, 17})

			Require(!GeometryBufferUpload::Copy(&buffer, 16, bytes, payload), "Invalid geometry copy size accepted");

		Require(!GeometryBufferUpload::Copy(&buffer, 16, 16, nullptr) && buffer.locks == 0, "Rejected geometry copy reached driver");

		DWORD bytes = 123;

		Require(!GeometryBufferUpload::ByteSize((std::numeric_limits<int>::max)(), 12, bytes) && bytes == 0, "Geometry size overflow accepted");

		report.Write("PASS controlled geometry: lock failure, null lock data, unlock failure, successful retries, copy bounds and allocation overflow");



		for (int cycle = 0; cycle < 16; ++cycle)

		{

			CGraphicVertexBuffer vertices;

			CGraphicIndexBuffer indices;

			void* data = reinterpret_cast<void*>(1);

			Require(!vertices.Lock(&data) && data == nullptr, "Empty vertex lock retained output");

			data = reinterpret_cast<void*>(1);

			Require(!indices.Lock(&data) && data == nullptr, "Empty index lock retained output");

			Require(!vertices.Create(-1, D3DFVF_XYZ, 0, D3DPOOL_DEFAULT) && vertices.GetVertexCount() == 0, "Negative vertex count accepted");

			Require(!vertices.Create((std::numeric_limits<int>::max)(), D3DFVF_XYZ, 0, D3DPOOL_DEFAULT), "Overflowing vertex allocation accepted");

			Require(!vertices.Create(3, D3DFVF_XYZ, 0, D3DPOOL_MANAGED), "Ex managed vertex pool accepted");

			Require(vertices.Create(3, D3DFVF_XYZ, D3DUSAGE_DYNAMIC | D3DUSAGE_WRITEONLY, D3DPOOL_DEFAULT), "Create combined-usage vertex buffer");

			const auto* original = vertices.GetD3DVertexBuffer();

			Require(!vertices.CreateDeviceObjects() && vertices.GetD3DVertexBuffer() == original, "Duplicate vertex allocation replaced live buffer");

			Require(!vertices.LockRange(4, &data) && data == nullptr, "Oversized vertex lock accepted");

			Require(vertices.LockRange(3, &data), "Combined-usage vertex write lock failed");

			memset(data, 0, 36);

			Require(vertices.Unlock(), "Vertex upload unlock failed");

			Require(!vertices.Copy(37, payload) && vertices.Copy(16, payload), "Vertex copy rejection/reuse failed");

			Require(!indices.Create(-1, D3DFMT_INDEX16) && !indices.Create(3, D3DFMT_UNKNOWN), "Invalid index descriptor accepted");

			const auto format = cycle % 2 ? D3DFMT_INDEX32 : D3DFMT_INDEX16;

			Require(indices.Create(4, format), "Index allocation failed");

			const int size = cycle % 2 ? 16 : 8;

			Require(!indices.Copy(size + 1, payload) && indices.Copy(size, payload), "Index copy rejection/reuse failed");

			D3DINDEXBUFFER_DESC desc = {};

			Check(indices.GetD3DIndexBuffer()->GetDesc(&desc), "Read index descriptor");

			Require(desc.Format == format && desc.Size == size, "Index allocation size/format mismatch");

			Require(!indices.CreateDeviceObjects(), "Duplicate index allocation accepted");

			Require(vertices.CreateDeviceObjects(), "Vertex descriptor lost during device-only release");

		}

	}



	void RunStartupCases(HWND window, Report& report)

	{

		report.Write("RUN startup: controlled capability/rollback decisions and production device creation");

		// Exercise the shared rollback on early exits at different partial stages.

		for (int failAfter = 0; failAfter < 5; ++failAfter)

		{

			int live = 0, cleanups = 0;

			const auto attempt = [&]()

			{

				const auto cleanup = [&]() { live = 0; ++cleanups; };

				GraphicsStartup::Rollback<decltype(cleanup)> rollback(cleanup);

				for (int step = 0; step < 5; ++step)

				{

					if (step == failAfter) return;

					++live;

				}

				rollback.Complete();

			};

			attempt();

			Require(live == 0 && cleanups == 1, "Partial startup not rolled back exactly once");

		}

		int cleanups = 0;

		const auto cleanup = [&]() { ++cleanups; };

		{

			GraphicsStartup::Rollback<decltype(cleanup)> rollback(cleanup);

			rollback.Complete();

		}

		Require(cleanups == 0, "Successful startup rolled back");

		try

		{

			GraphicsStartup::Rollback<decltype(cleanup)> rollback(cleanup);

			throw std::runtime_error("controlled startup unwind");

		}

		catch (const std::runtime_error&) {}

		Require(cleanups == 1, "Exceptional startup not rolled back");



		for (int scenario = 0; scenario < 5; ++scenario)

		{

			int queries = 0;

			const auto samples = GraphicsStartup::SelectMultisampling(scenario != 4,

				D3DFMT_X8R8G8B8, D3DFMT_D24S8,

				[&](D3DFORMAT format, D3DMULTISAMPLE_TYPE type, DWORD* levels) -> HRESULT

				{

					++queries;

					*levels = 1;

					if (format == D3DFMT_D24S8)

					{

						if (scenario == 1 && type == D3DMULTISAMPLE_4_SAMPLES) return D3DERR_NOTAVAILABLE;

						if (scenario == 2) return E_FAIL;

						if (scenario == 3) *levels = 0;

					}

					return S_OK;

				});

			const auto expected = scenario == 0 ? D3DMULTISAMPLE_4_SAMPLES :

				(scenario == 1 ? D3DMULTISAMPLE_2_SAMPLES : D3DMULTISAMPLE_NONE);

			Require(samples == expected, "Color/depth MSAA selection mismatch");

			if (scenario == 4) Require(queries == 0, "Disabled MSAA queried hardware");

		}

		Require(GraphicsStartup::BestFilter(0, D3DPTFILTERCAPS_MAGFANISOTROPIC, D3DPTFILTERCAPS_MAGFLINEAR) == D3DTEXF_POINT &&

			GraphicsStartup::BestFilter(D3DPTFILTERCAPS_MAGFLINEAR, D3DPTFILTERCAPS_MAGFANISOTROPIC, D3DPTFILTERCAPS_MAGFLINEAR) == D3DTEXF_LINEAR &&

			GraphicsStartup::BestFilter(D3DPTFILTERCAPS_MAGFANISOTROPIC, D3DPTFILTERCAPS_MAGFANISOTROPIC, D3DPTFILTERCAPS_MAGFLINEAR) == D3DTEXF_ANISOTROPIC,

			"Capability filter fallback mismatch");

		report.Write("PASS controlled startup: partial/success/exception rollback, paired MSAA fallback, filter capability choices");



		class StartupDevice : public CGraphicDevice

		{

		public:

			bool Empty() const

			{

				if (ms_lpd3d || ms_lpd3dDevice || ms_lpd3dMatStack || m_pStateManager || ms_hDC ||

					ms_ptVS || ms_pntVS || ms_pnt2VS || ms_lpSphereMesh || ms_lpCylinderMesh) return false;

				for (auto* buffer : ms_alpd3dDefIB) if (buffer) return false;

				for (auto* buffer : ms_alpd3dPDTVB) if (buffer) return false;

				return true;

			}

			void Verify()

			{

				Require(ms_lpd3dDevice && ms_lpd3dMatStack && m_pStateManager && ms_ptVS && ms_pntVS && ms_pnt2VS,

					"Required startup resource absent");

				Check(m_pStateManager->GetDefaultStateResult(), "Production default state");

				for (auto* buffer : ms_alpd3dDefIB) Require(buffer != nullptr, "Default index buffer absent");

				for (auto* buffer : ms_alpd3dPDTVB) Require(buffer != nullptr, "Dynamic vertex buffer absent");

				D3DDEVICE_CREATION_PARAMETERS creation = {};

				Check(ms_lpd3dDevice->GetCreationParameters(&creation), "Read production processing mode");

				const bool softwareOnly = (creation.BehaviorFlags & D3DCREATE_SOFTWARE_VERTEXPROCESSING) != 0;

				Require((ms_lpd3dDevice->GetSoftwareVertexProcessing() != FALSE) == softwareOnly, "Default processing mode mismatch");

				Check(m_pStateManager->BeginScene(), "Production startup BeginScene");

				Check(m_pStateManager->EndScene(), "Production startup EndScene");

			}

		};

		// Run before TestDevice owns the shared CGraphicBase pointers.

		StartupDevice owner;

		owner.InitBackBufferCount(2);

		Require(owner.Create(window, 0, 64, true) == CGraphicDevice::CREATE_DEVICE && owner.Empty(), "Invalid startup did not clean up");

		for (int cycle = 0; cycle < 2; ++cycle)

		{

			Require((owner.Create(window, 64, 64, true) & CGraphicDevice::CREATE_OK) != 0, "Production device startup failed; see Graphics diagnostics");

			owner.Verify();

			owner.Destroy();

			Require(owner.Empty(), "Production teardown retained owned resources");

		}

		report.Write("PASS production startup: invalid-size rejection/reuse, two create/default-state/resource/scene/destroy cycles, empty owned pointers after teardown");

		report.Write("Startup limits: windowed HAL only; rollback failures simulated, no real allocation failure injection; optional legacy helper meshes may be unavailable");

	}

			}

			report.Write("RUN expected rejection: %s (uploader error messages are expected)", invalidNames[index]);

			CGraphicImageTexture texture;

			texture.SetFileName(invalidNames[index]);

			Require(!texture.CreateFromDecodedData(invalid, format, D3DX_FILTER_NONE), "Invalid input was accepted");

			Require(texture.IsEmpty() && !texture.GetD3DTexture() &&

				texture.GetWidth() == 0 && texture.GetHeight() == 0, "Rejected input published texture state");

			UploadAndVerify(device, texture, Pattern(2, 2, index), D3DFMT_A8R8G8B8, report);

			report.Write("PASS rejection and subsequent reuse: %s", invalidNames[index]);

		}



		for (unsigned iteration = 0; iteration < 64; ++iteration)

		{

			report.Write("RUN creation/release cycle %u/64", iteration + 1);

			const int size = (iteration % 2) ? 32 : 8;

			CGraphicImageTexture texture;

			UploadAndVerify(device, texture, Pattern(size, size, iteration),

				(iteration % 2) ? D3DFMT_UNKNOWN : D3DFMT_A8R8G8B8, report);

		}

		RunBlockedDrawCases(device, report);

		report.Write("Coverage: %d/7 size cases, 12 rejection/reuse cases, 64 creation/release cycles", 7 - skipped);

		report.Write("Limits: failures/padding are forced in the CPU backend; no real driver faults, full leak proof, recovery or performance test.");

		report.Write("GPU cases use driver-selected upload pitch; logged pitch belongs to readback surfaces.");

		return (skipped || !shaderConstantsPassed) ? 3 : 0;

	}

	}

	catch (...)

	{

		report.Write("FAIL: unexpected C++ exception");

	}

	// RunCases has destroyed all test resources and the device before this result.
